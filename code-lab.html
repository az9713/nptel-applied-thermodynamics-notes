    """Check the mixer, upstream throttle, and saturated-liquid discharge."""

from math import isclose, log


CP_AIR = 1.005  # kJ/(kg K), constant teaching model
R_AIR = 0.287  # kJ/(kg K)


def entropy_change(t_initial, p_initial, t_final, p_final):
    """Return ideal-gas entropy change in kJ/(kg K); use K and kPa."""
    return CP_AIR * log(t_final / t_initial) - R_AIR * log(
        p_final / p_initial
    )


def check_draining_tank():
    pressure = 2000.0  # kPa
    tank_volume, initial_mass, reservoir_temperature = 0.600, 8.0, 523.15
    # IF97 saturation properties at 2 MPa, with h in kJ/kg and v in m³/kg.
    v_f, v_g = 0.00117675024425, 0.09958054416890
    h_f, h_g, t_sat = 908.62185113646, 2798.38414024152, 485.53453531849
    x_initial = (tank_volume / initial_mass - v_f) / (v_g - v_f)
    final_mass = tank_volume / v_g
    vapor_gain = initial_mass * (1 - x_initial) * v_f / v_g
    heat = vapor_gain * (h_g - h_f)
    u_initial = h_f + x_initial * (h_g - h_f)
    u_initial -= pressure * tank_volume / initial_mass
    u_final = h_g - pressure * v_g
    outflow = initial_mass - final_mass
    full_balance = final_mass * u_final - initial_mass * u_initial
    full_balance += outflow * h_f
    assert isclose(heat, full_balance, rel_tol=1e-10)
    generation = heat * (1 / t_sat - 1 / reservoir_temperature)
    assert 0 < generation
    print(f"Draining tank: {outflow:.5f} kg out, {heat:.2f} kJ in, "
          f"{1000 * generation:.2f} J/K generated")


def main():
    hot_temperature, cold_temperature = 400.0, 300.0
    hot_flow, cold_flow = 1.0, 1.0  # kg/s
    chamber_pressure, upstream_pressure = 100.0, 200.0  # kPa

    outlet_flow = hot_flow + cold_flow
    outlet_temperature = (
        hot_flow * hot_temperature + cold_flow * cold_temperature
    ) / outlet_flow
    chamber_generation = hot_flow * entropy_change(
        hot_temperature, chamber_pressure, outlet_temperature,
        chamber_pressure,
    ) + cold_flow * entropy_change(
        cold_temperature, chamber_pressure, outlet_temperature,
        chamber_pressure,
    )
    throttle_generation = hot_flow * entropy_change(
        hot_temperature, upstream_pressure, hot_temperature,
        chamber_pressure,
    )
    combined_generation = hot_flow * entropy_change(
        hot_temperature, upstream_pressure, outlet_temperature,
        chamber_pressure,
    ) + cold_flow * entropy_change(
        cold_temperature, chamber_pressure, outlet_temperature,
        chamber_pressure,
    )

    thermal_enthalpy_residual = CP_AIR * (
        outlet_flow * outlet_temperature
        - hot_flow * hot_temperature - cold_flow * cold_temperature
    )
    assert isclose(thermal_enthalpy_residual, 0.0, abs_tol=1e-12)
    assert isclose(
        combined_generation, chamber_generation + throttle_generation,
        rel_tol=1e-12,
    )
    assert 0 < chamber_generation < combined_generation
    assert entropy_change(350.0, 100.0, 350.0, 100.0) == 0

    print(f"Outlet temperature: {outlet_temperature:.1f} K")
    print(f"Mixer entropy generation: {chamber_generation:.5f} kW/K")
    print(f"Throttle entropy generation: {throttle_generation:.5f} kW/K")
    print(f"Combined entropy generation: {combined_generation:.5f} kW/K")
    check_draining_tank()


if __name__ == "__main__":
    main()

    """Compare reversible ideal-gas compression paths at fixed end pressures."""

from math import isclose, log


def compressor_paths(t_in, pressure_ratio, gas_constant, cp):
    """Return (isothermal input, isentropic input, isentropic outlet T).

    Temperatures are in kelvin and work and heat capacities are mass specific,
    using kJ/kg and kJ/(kg K). Work input is reported as a positive magnitude.
    The gas is ideal with constant cp, and both paths are internally reversible.
    """
    if t_in <= 0 or pressure_ratio < 1 or not 0 < gas_constant < cp:
        raise ValueError("Require T > 0, pressure ratio >= 1, and cp > R > 0")
    t_out_s = t_in * pressure_ratio ** (gas_constant / cp)
    isothermal = gas_constant * t_in * log(pressure_ratio)
    isentropic = cp * (t_out_s - t_in)
    return isothermal, isentropic, t_out_s


def main():
    t_in, pressure_ratio, gas_constant, cp = 300, 8, 0.287, 1.005
    isothermal, isentropic, t_out_s = compressor_paths(
        t_in, pressure_ratio, gas_constant, cp
    )
    entropy_rise = cp * log(t_out_s / t_in)
    entropy_rise -= gas_constant * log(pressure_ratio)
    assert isclose(entropy_rise, 0, abs_tol=1e-12)
    assert 0 < isothermal < isentropic
    assert compressor_paths(t_in, 1, gas_constant, cp) == (0, 0, t_in)
    print(f"Isothermal shaft input: {isothermal:.2f} kJ/kg")
    print(f"Isentropic shaft input: {isentropic:.2f} kJ/kg")
    print(f"Isentropic outlet temperature: {t_out_s:.2f} K")


if __name__ == "__main__":
    main()

from math import isclose, log, pi, sqrt


def check_vortex_tube():
    gas_constant = 287.0
    cp = 1.4 * gas_constant / (1.4 - 1)
    t_in, t_hot = 293.15, 389.15
    c_in, c_hot, c_cold = 1.0, 100.0, 200.0
    t_cold = 2 * t_in - t_hot
    t_cold += (c_in**2 - (c_hot**2 + c_cold**2) / 2) / cp
    energy_in = cp * t_in + c_in**2 / 2
    energy_out = cp * (t_hot + t_cold) / 2 + (c_hot**2 + c_cold**2) / 4
    assert isclose(energy_in, energy_out, rel_tol=1e-12)
    pressure_ratio = 101.3 / 1000
    generation = (
        cp * (log(t_hot / t_in) + log(t_cold / t_in)) / 2
        - gas_constant * log(pressure_ratio)
    )
    assert generation > 0
    print(f"Vortex tube cold exit: {t_cold:.2f} K, {generation:.2f} W/K generated")


def main():
    cp, gas_constant = 1010.0, 287.0
    area, outlet_pressure = pi * 0.10**2 / 4, 400_000
    streams = [(6.0, 473.15, 600_000, 100), (1.0, 373.15, 500_000, 50)]
    flow = sum(mass for mass, *_ in streams)
    energy = sum(
        mass * (cp * temp + speed**2 / 2)
        for mass, temp, _, speed in streams
    ) / flow
    velocity_factor = flow * gas_constant / (outlet_pressure * area)
    root = sqrt(cp**2 + 2 * velocity_factor**2 * energy)
    temperature = 2 * energy / (cp + root)
    speed = velocity_factor * temperature
    generation = sum(
        mass * (cp * log(temperature / temp)
                - gas_constant * log(outlet_pressure / pressure))
        for mass, temp, pressure, _ in streams
    )
    density = outlet_pressure / (gas_constant * temperature)
    assert isclose(density * area * speed, flow, rel_tol=1e-12)
    assert isclose(cp * temperature + speed**2 / 2, energy, rel_tol=1e-12)
    assert generation > 0
    print(f"Outlet: {temperature:.2f} K, {speed:.2f} m/s")
    print(f"Generation: {generation:.2f} W/K")
    check_vortex_tube()


if __name__ == "__main__":
    main()

from math import isclose, log, log1p


def thermal_exergy(heat_capacity, temperature, ambient):
    """Use total heat capacity in kJ/K and absolute temperatures in K."""
    relative_gap = (temperature - ambient) / ambient
    return heat_capacity * ambient * (relative_gap - log1p(relative_gap))


def pressure_exergy(mass, gas_constant, temperature, pressure, ambient):
    """Compute physical exergy of ideal gas at ambient temperature."""
    ratio = pressure / ambient
    return mass * gas_constant * temperature * (log(ratio) + 1 / ratio - 1)


def main():
    hot = thermal_exergy(4.18, 350.0, 300.0)
    cold = thermal_exergy(4.18, 250.0, 300.0)
    assert cold > hot > 0
    assert thermal_exergy(4.18, 300.0, 300.0) == 0
    reduced = pressure_exergy(1.0, 0.287, 300.0, 200.0, 100.0)
    volume_change = 0.287 * 300 * (1 / 200 - 1 / 100)
    entropy_change = -0.287 * log(2)
    assert isclose(reduced, 100 * volume_change - 300 * entropy_change)
    print(f"Thermal availability: hot {hot:.3f} kJ, cold {cold:.3f} kJ")
    print(f"Compressed 1 kg ideal air: {reduced:.3f} kJ")

from math import isclose, log


def main():
    cp, r = 1.005, 0.287
    t0, p0 = 300.0, 100.0
    outlet_t, outlet_p = 450.0, 400.0
    flow, shaft_input = 0.04, 8.0
    enthalpy_delta = cp * (outlet_t - t0)
    entropy_delta = cp * log(outlet_t / t0) - r * log(outlet_p / p0)
    heat_into_device = flow * enthalpy_delta - shaft_input
    generation = flow * entropy_delta - heat_into_device / t0
    product = flow * (enthalpy_delta - t0 * entropy_delta)
    destruction = shaft_input - product
    assert isclose(destruction, t0 * generation, rel_tol=1e-12)
    assert 0 < product < shaft_input
    engine_resource = 100 * (1 - 300 / 500)
    assert isclose(30 / engine_resource, 0.75)
    print(f"Compressor exergy gain: {product:.4f} kW")
    print(f"Compressor exergy destruction: {destruction:.4f} kW")
    print(f"Compressor second-law efficiency: {product/shaft_input:.2%}")
    print(f"Heat engine second-law efficiency: {30/engine_resource:.2%}")

from math import isclose, log


def main():
    r, gamma = 0.287, 1.4
    cv = r / (gamma - 1)
    cp = cv + r
    p0, t0 = 100.0, 298.15
    p1, t1, volume1 = p0, t0, 0.002
    p2, t2, useful_input = 600.0, 423.15, 1.2
    mass = p1 * volume1 / (r * t1)
    volume2 = mass * r * t2 / p2
    delta_u = mass * cv * (t2 - t1)
    delta_s = mass * (cp * log(t2 / t1) - r * log(p2 / p1))
    delta_b = delta_u + p0 * (volume2 - volume1) - t0 * delta_s
    destruction = useful_input - delta_b
    assert 0 < delta_b < useful_input

    capacity_hot, capacity_cold = 24.0, 418.0
    t_hot, t_cold, work = 358.15, 293.15, 240.0
    final = (
        capacity_hot * t_hot + capacity_cold * t_cold + work
    ) / (capacity_hot + capacity_cold)
    assert isclose(capacity_hot * (final - t_hot)
                   + capacity_cold * (final - t_cold), work)
    generation = capacity_hot * log(final / t_hot)
    generation += capacity_cold * log(final / t_cold)
    assert generation > 0
    print(f"Compression exergy change: {delta_b:.4f} kJ")
    print(f"Compression destruction: {destruction:.4f} kJ")
    print(f"Stirred tank: {final:.3f} K, {generation:.5f} kJ/K")


if __name__ == "__main__":
    main()

from math import isclose, log


def main():
    hot_capacity_rate, cold_capacity_rate = 1.9, 2.1
    hot_in, cold_in, cold_out = 520.0, 300.0, 390.0
    ambient = 300.0
    heating_rate = cold_capacity_rate * (cold_out - cold_in)
    hot_out = hot_in - heating_rate / hot_capacity_rate
    assert hot_in > cold_out and hot_out > cold_in
    hot_entropy = hot_capacity_rate * log(hot_out / hot_in)
    cold_entropy = cold_capacity_rate * log(cold_out / cold_in)
    hot_resource = heating_rate + ambient * hot_entropy
    cold_product = heating_rate - ambient * cold_entropy
    destruction = ambient * (hot_entropy + cold_entropy)
    assert 0 < cold_product < hot_resource
    assert isclose(hot_resource - cold_product, destruction)

    fusion_heat, freezing_temperature = 334.0, 273.15
    exergy_gain = -fusion_heat + ambient * fusion_heat / freezing_temperature
    assert exergy_gain > 0
    print(f"Heat exchanger hot outlet: {hot_out:.2f} K")
    print(f"Heat exchanger destruction: {destruction:.3f} kW")
    print(f"Freezing exergy gain: {exergy_gain:.3f} kJ/kg")

from math import isclose


def rankine(h_liquid, h_pump_out, h_turbine_in, h_turbine_out):
    """Return positive net work, boiler heat and efficiency using kJ/kg throughout."""
    if not h_liquid < h_pump_out < h_turbine_in or not h_liquid < h_turbine_out < h_turbine_in:
        raise ValueError("Expected physically ordered Rankine enthalpies")
    work = (h_turbine_in - h_turbine_out) - (h_pump_out - h_liquid)
    if work <= 0:
        raise ValueError("Expansion must deliver more work than pumping consumes")
    heat_in = h_turbine_in - h_pump_out
    heat_out = h_turbine_out - h_liquid
    assert isclose(work, heat_in - heat_out)
    return work, heat_in, work / heat_in


def feedwater_bleed(h_cold, h_mixed, h_bleed):
    """Return extracted vapor mass per unit mass leaving an open feedwater heater."""
    if not h_cold < h_mixed < h_bleed:
        raise ValueError("Expected cold < mixed < bleed enthalpy")
    return (h_mixed - h_cold) / (h_bleed - h_cold)


if __name__ == "__main__":
    net, heat, efficiency = rankine(191.81, 199.87, 3349.53, 2109.33)
    bleed = feedwater_bleed(192.31, 640.09, 2650.0)
    assert isclose((1 - bleed) * 192.31 + bleed * 2650.0, 640.09)
    assert 0 < efficiency < 1
    try:
        rankine(0, 200, 250, 249)
    except ValueError:
        pass
    else:
        raise AssertionError("Rejected heat exceeds added heat in the negative-work case")
    print(f"Net output {net:.2f} kJ/kg; input {heat:.2f} kJ/kg; efficiency {efficiency:.2%}")
    print(f"Example open-heater bleed fraction: {bleed:.2%}")

    Download Python source

    View code
    """Sweep gas-turbine pressure ratio at fixed cold and hot temperatures."""

from math import isclose, sqrt


def brayton(t_cold, t_hot, ratio, eta_c=1.0, eta_t=1.0, cp=1.005, gamma=1.4):
    """Return compressor and turbine exits, net work, and thermal efficiency."""
    if not (0 < t_cold < t_hot and ratio > 1 and cp > 0 and gamma > 1):
        raise ValueError("Require a positive temperature range and pressure ratio > 1")
    if not (0 < eta_c <= 1 and 0 < eta_t <= 1):
        raise ValueError("Device efficiencies must lie in (0, 1]")
    a = (gamma - 1) / gamma
    scale = ratio**a
    t_out_c = t_cold + t_cold * (scale - 1) / eta_c
    t_out_t = t_hot - eta_t * t_hot * (1 - 1 / scale)
    heat_in = cp * (t_hot - t_out_c)
    heat_out = cp * (t_out_t - t_cold)
    work = heat_in - heat_out
    if heat_in <= 0 or work <= 0:
        raise ValueError("The specified cycle does not operate as a heat engine")
    return t_out_c, t_out_t, work, work / heat_in


if __name__ == "__main__":
    cold, hot = 300, 1500
    optimum = sqrt(hot / cold) ** (1.4 / 0.4)
    best = brayton(cold, hot, optimum)
    assert isclose(best[0], best[1])
    assert best[2] > max(brayton(cold, hot, r)[2] for r in (0.93 * optimum, 1.07 * optimum))
    real = brayton(cold, hot, 10, 0.88, 0.91)
    assert real[2] < brayton(cold, hot, 10)[2]
    print(f"Ideal max work: {best[2]:.2f} kJ/kg at pressure ratio {optimum:.3f}")
    print(f"At pressure ratio 10 with nonideal components: {real[2]:.2f} kJ/kg, {real[3]:.2%}")

    Download Python source

    """Compute a two-stage ideal Brayton cycle with optional heat recovery."""

from math import isclose


def modified_brayton(cold, hot, ratio, epsilon, cp=1.005, gamma=1.4):
    """Return net work, external heat supplied, recovered heat and efficiency."""
    if not (0 < cold < hot and ratio > 1 and 0 <= epsilon <= 1 and cp > 0 and gamma > 1):
        raise ValueError("Invalid temperatures, pressure ratio, effectiveness or gas properties")
    stage_scale = ratio ** ((gamma - 1) / (2 * gamma))
    after_compression = cold * stage_scale
    after_expansion = hot / stage_scale
    recovered = epsilon * cp * max(0, after_expansion - after_compression)
    heat_in = cp * (hot - after_compression + hot - after_expansion) - recovered
    heat_out = cp * (after_compression - cold + after_expansion - cold) - recovered
    net_work = 2 * cp * ((hot - after_expansion) - (after_compression - cold))
    if heat_in <= 0 or net_work <= 0:
        raise ValueError("No positive power-cycle solution")
    assert isclose(heat_in - heat_out, net_work)
    return net_work, heat_in, recovered, net_work / heat_in


if __name__ == "__main__":
    for epsilon in (0, 0.4, 0.8):
        net, heat, recovery, efficiency = modified_brayton(300, 1500, 10, epsilon)
        print(f"Effectiveness {epsilon:.1f}: work={net:.1f}, heat={heat:.1f}, recovered={recovery:.1f} kJ/kg, efficiency={efficiency:.2%}")
    off = modified_brayton(300, 1500, 10, 0)
    on = modified_brayton(300, 1500, 10, 0.8)
    assert isclose(off[0], on[0]) and isclose(off[1] - on[1], on[2])
    assert on[-1] > off[-1]

from math import isclose


def reciprocating(kind, t_intake, t_peak, compression, cv=0.718, gamma=1.4):
    """Return heat received, net work, efficiency, and MEP divided by intake pressure."""
    if kind not in ("Otto", "Diesel"):
        raise ValueError("Cycle must be Otto or Diesel")
    if not (0 < t_intake < t_peak and compression > 1 and cv > 0 and gamma > 1):
        raise ValueError("Invalid thermal or geometric inputs")
    a = gamma - 1
    t_compressed = t_intake * compression**a
    cutoff = t_peak / t_compressed if kind == "Diesel" else 1
    if t_peak <= t_compressed or cutoff >= compression:
        raise ValueError("Heat addition and expansion both require positive ranges")
    t_expanded = t_peak * (cutoff / compression) ** a
    c_heat = cv * gamma if kind == "Diesel" else cv
    heat_in = c_heat * (t_peak - t_compressed)
    work = heat_in - cv * (t_expanded - t_intake)
    gas_constant = cv * a
    mep_ratio = work / (gas_constant * t_intake * (1 - 1 / compression))
    return heat_in, work, work / heat_in, mep_ratio


if __name__ == "__main__":
    otto = reciprocating("Otto", 300, 1800, 10)
    diesel = reciprocating("Diesel", 300, 1800, 10)
    diesel_high = reciprocating("Diesel", 300, 1800, 18)
    assert isclose(otto[2], 1 - 10**-0.4)
    assert diesel[2] < otto[2] < diesel_high[2]
    for label, (q_in, work, eta, mep_ratio) in zip(("Otto r=10", "Diesel r=10", "Diesel r=18"), (otto, diesel, diesel_high)):
        print(f"{label:12}: heat={q_in:.1f}, work={work:.1f} kJ/kg, efficiency={eta:.2%}, MEP={100*mep_ratio:.1f} kPa at 100 kPa intake")

from math import isclose


def refrigeration(h_suction, h_discharge_s, h_liquid, efficiency, load):
    """Return fluid flow, compressor power, rejection rate and COP in kg/s and kW."""
    if not (h_discharge_s > h_suction > h_liquid and 0 < efficiency <= 1 and load > 0):
        raise ValueError("Need positive load, refrigeration effect and compression work")
    h_discharge = h_suction + (h_discharge_s - h_suction) / efficiency
    flow = load / (h_suction - h_liquid)
    power = flow * (h_discharge - h_suction)
    rejected = flow * (h_discharge - h_liquid)
    assert isclose(rejected, load + power)
    return flow, power, rejected, load / power


if __name__ == "__main__":
    base = refrigeration(395.66, 419.56, 241.72, 0.75, 10)
    subcooled = refrigeration(395.66, 419.56, 233.13, 0.75, 10)
    ideal = refrigeration(395.66, 419.56, 241.72, 1, 10)
    assert subcooled[0] < base[0] and base[-1] < subcooled[-1] < ideal[-1]
    print(f"Base: flow={base[0]:.5f} kg/s, power={base[1]:.3f} kW, heat rejected={base[2]:.3f} kW, COP={base[3]:.3f}")
    print(f"Subcooled: flow={subcooled[0]:.5f} kg/s, power={subcooled[1]:.3f} kW, COP={subcooled[3]:.3f}")

from math import exp, isclose


def saturation_pressure(temperature_c):
    """IAPWS SR1-86(1992) saturation pressure above liquid water, in kPa."""
    temperature = temperature_c + 273.15
    critical = 647.096
    if not 0.01 <= temperature_c <= critical - 273.15:
        raise ValueError("Temperature outside the liquid-vapor saturation curve")
    theta = 1 - temperature / critical
    coefficients = (-7.85951783, 1.84408259, -11.7866497, 22.6807411, -15.9618719, 1.80122502)
    powers = (1, 1.5, 3, 3.5, 4, 7.5)
    logarithm = (critical / temperature) * sum(
        coefficient * theta**power for coefficient, power in zip(coefficients, powers)
    )
    return 22064 * exp(logarithm)


def dew_point(vapor_pressure):
    low, high = 0.01, 60.0
    if not saturation_pressure(low) <= vapor_pressure <= saturation_pressure(high):
        raise ValueError("Vapor pressure outside this ambient-range dew-point calculation")
    for _ in range(50):
        middle = (low + high) / 2
        if saturation_pressure(middle) < vapor_pressure:
            low = middle
        else:
            high = middle
    return (low + high) / 2

from math import isclose, sqrt


def stagnation(temperature, pressure, speed, gamma=1.4, gas_constant=287.0):
    """Return acoustic speed, Mach number, total temperature and pressure.

    Inputs use K, Pa, m/s and J/(kg K). The returned pressure is in Pa.
    The gas is calorically perfect and the hypothetical stop is isentropic.
    """
    if temperature <= 0 or pressure <= 0 or gamma <= 1 or gas_constant <= 0:
        raise ValueError("Require positive T, p, R, and gamma > 1")
    sound_speed = sqrt(gamma * gas_constant * temperature)
    mach = abs(speed) / sound_speed
    temperature_ratio = 1 + (gamma - 1) * mach**2 / 2
    return (
        sound_speed, mach, temperature * temperature_ratio,
        pressure * temperature_ratio ** (gamma / (gamma - 1))
    )


def main():
    a, mach, total_t, total_p = stagnation(300.0, 100_000.0, 250.0)
    cp = 1.4 * 287.0 / (1.4 - 1)
    assert isclose(cp * (total_t - 300.0), 250.0**2 / 2)
    assert stagnation(300.0, 100_000.0, 0)[1:] == (0, 300, 100_000)
    assert stagnation(300.0, 100_000.0, -250.0)[1:] == (
        mach, total_t, total_p
    )
    print(f"Sound speed: {a:.2f} m/s; Mach: {mach:.4f}")
    print(f"Stagnation state: {total_t:.2f} K, {total_p/1000:.2f} kPa")
    _, m2, t2, p2 = stagnation(300.0, 100_000.0, 150.0)

from math import isclose, log, sqrt


def normal_shock(mach_upstream, gamma=1.4):
    """Return downstream Mach and downstream/upstream p, rho, T and p_t."""
    if mach_upstream < 1 or gamma <= 1:
        raise ValueError("Require M >= 1 and gamma > 1")
    mach_sq = mach_upstream**2
    pressure_ratio = 1 + 2 * gamma / (gamma + 1) * (mach_sq - 1)
    density_ratio = (gamma + 1) * mach_sq / ((gamma - 1) * mach_sq + 2)
    temperature_ratio = pressure_ratio / density_ratio
    mach_downstream = mach_upstream / density_ratio / sqrt(temperature_ratio)
    factor = (gamma - 1) / 2
    total_ratio = pressure_ratio * (
        (1 + factor * mach_downstream**2) / (1 + factor * mach_sq)
    ) ** (gamma / (gamma - 1))
    return (
        mach_downstream, pressure_ratio, density_ratio, temperature_ratio,
        total_ratio
    )


def main():
    gamma, gas_constant, mach = 1.4, 287.0, 2.0
    mach2, p2, rho2, t2, pt2 = normal_shock(mach, gamma)
    assert normal_shock(1, gamma) == (1, 1, 1, 1, 1)
    assert isclose(1 + gamma * mach**2, p2 * (1 + gamma * mach2**2))
    assert isclose(1 + (gamma - 1) / 2 * mach**2,
                   t2 * (1 + (gamma - 1) / 2 * mach2**2))
    assert 0 < pt2 < 1
    print(f"M2={mach2:.5f}, p2/p1={p2:.5f}, rho2/rho1={rho2:.5f}")

from math import isclose, sqrt


def area_ratio(mach, gamma=1.4):
    """Return A/A* for positive Mach in a calorically perfect isentrope."""
    if mach <= 0 or gamma <= 1:
        raise ValueError("Require M > 0 and gamma > 1")
    factor = (2 + (gamma - 1) * mach**2) / (gamma + 1)
    return factor ** ((gamma + 1) / (2 * (gamma - 1))) / mach


def mach_from_area(ratio, supersonic, gamma=1.4):
    """Find the selected Mach root by bisection."""
    if ratio < 1 or gamma <= 1:
        raise ValueError("Require A/A* >= 1 and gamma > 1")
    if ratio == 1:
        return 1.0

    if supersonic:
        lo, hi = 1.0, 2.0
        while area_ratio(hi, gamma) < ratio:
            hi *= 2
    else:
        lo, hi = 0.5, 1.0
        while area_ratio(lo, gamma) < ratio:
            lo /= 2

    for _ in range(70):
        mid = (lo + hi) / 2
        area_is_high = area_ratio(mid, gamma) > ratio
        if area_is_high == supersonic:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2


def pressure_ratio(mach, gamma=1.4):
    """Return local static pressure divided by local stagnation pressure."""
    return (1 + (gamma - 1) * mach**2 / 2) ** (-gamma / (gamma - 1))


def sonic_flux(total_pressure, total_temperature, gamma=1.4,
               gas_constant=287.0):
    """Return choked mass flow per unit area in kg/(m² s), using Pa and K."""
    if total_pressure <= 0 or total_temperature <= 0:
        raise ValueError("Require positive total pressure and temperature")
    factor = (2 / (gamma + 1)) ** ((gamma + 1) / (2 * (gamma - 1)))
    return (
        total_pressure / sqrt(total_temperature)
        * sqrt(gamma / gas_constant) * factor
    )


def main():
    ratio = 2.0
    mach_sub = mach_from_area(ratio, False)
    mach_sup = mach_from_area(ratio, True)
    assert mach_sub < 1 < mach_sup
    assert isclose(area_ratio(mach_sub), ratio, rel_tol=1e-12)
    assert isclose(area_ratio(mach_sup), ratio, rel_tol=1e-12)
    assert mach_from_area(1, False) == mach_from_area(1, True) == 1
    print(f"Roots at A/A*=2: {mach_sub:.5f}, {mach_sup:.5f}")

from math import isclose, sqrt

def trial_shock(area_to_throat, exit_to_throat, gamma=1.4):
    """Return exit p/p_t1, exit Mach and upstream Mach at a trial shock."""
    if not 1 <= area_to_throat <= exit_to_throat:
        raise ValueError("Shock must lie between the throat and exit")
    mach_a = mach_from_area(area_to_throat, True, gamma)
    shock = normal_shock(mach_a, gamma)
    total_ratio = shock[-1]
    mach_exit = mach_from_area(max(1.0, exit_to_throat * total_ratio),
                               False, gamma)
    return total_ratio * pressure_ratio(mach_exit, gamma), mach_exit, mach_a


def locate_shock(back_to_total, exit_to_throat, gamma=1.4):
    """Invert the post-shock exit pressure for a single interior normal shock."""
    lo, hi = 1.0, exit_to_throat
    high_back = trial_shock(lo, exit_to_throat, gamma)[0]
    low_back = trial_shock(hi, exit_to_throat, gamma)[0]
    if not low_back < back_to_total < high_back:
        raise ValueError("Back pressure does not select an interior shock")
    for _ in range(60):
        mid = (lo + hi) / 2
        if trial_shock(mid, exit_to_throat, gamma)[0] > back_to_total:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2


def steam_exit(p_total, v_total, exponent, throat_area, p_back):
    """Power-law steam expansion, p in Pa and v in m³/kg."""
    if not (0 < p_back <= p_total and v_total > 0 and exponent > 1
            and throat_area > 0):
        raise ValueError("Inputs outside the pressure or steam model domain")
    ratio_star = (2 / (exponent + 1)) ** (exponent / (exponent - 1))
    ratio = max(p_back / p_total, ratio_star)
    v = v_total * ratio ** (-1 / exponent)
    delta_h = exponent / (exponent - 1) * p_total * v_total * (
        1 - ratio ** ((exponent - 1) / exponent)
    )
    speed = sqrt(2 * delta_h)
    return p_total * ratio, speed, throat_area * speed / v
